    // ------------------------------------------------------------------ phòng khoá (module chung NekoGuard)
    Image guardImg;
    int guardFor;

    void paintGuard(Graphics gr, int w, int h) {
        NekoGuard.original = "Scott Cawthon";
        NekoGuard.en = FnafSave.lang == 1;
        NekoGuard.trip("FnafPack");
        if (guardImg == null || guardFor != w * 10000 + h) {
            guardFor = w * 10000 + h;
            guardImg = null;
            int[] hs = {96, 128, 176, 240, 320};
            int target = w > h ? h * 78 / 100 : h * 50 / 100, pick = 0;
            for (int i = 0; i < hs.length; i++) if (hs[i] <= target) pick = hs[i];
            if (pick > 0) {
                try {
                    // ảnh nằm ngoài data.bin (thư mục /ny của jar) để vẫn hiện khi gói tài nguyên bị sửa
                    guardImg = Image.createImage("/ny/nekoyuki_halfbody_" + pick + ".png");
                } catch (Throwable t) {
                    guardImg = null;
                }
            }
        }
        // màn lớn (J2ME Loader, APK): font hệ thống của Android đủ dấu và đủ to; màn nhỏ: font bitmap của game
        boolean big = Math.min(w, h) >= 360;
        NekoGuard.paint(gr, w, h, guardImg == null ? null : new Image[]{guardImg}, big ? null : guardText(fL), big ? null : guardText(fS),
                System.currentTimeMillis());
    }

    NekoGuard.Text guardText(final FnafFont f) {
        return new NekoGuard.Text() {
            public int height() {
                return f.getHeight() + 2;
            }

            public int width(String s) {
                return f.stringWidth(s);
            }

            public void draw(Graphics g, String s, int x, int y, int rgb) {
                f.drawRaw(g, s, x, y, rgb);
            }
        };
    }
